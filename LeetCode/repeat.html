    实现一个函数，间隔输出

    # 实现一个函数，间隔输出

    // 方式一
/**
 * @param fn 要间隔执行的函数
 * @param repeat 执行次数
 * @param interval 间隔时间(s)
 */
function createRepeat(fn, repeat, interval) {
	return async function (...args) {
		while (repeat) {
			await new Promise(resolve => {
				setTimeout(() => {
					repeat--
					fn.apply(this, args)
					resolve()
				}, interval * 1000)
			})
		}
	}
}

// 方式二
/**
 * @param fn 要间隔执行的函数
 * @param repeat 执行次数
 * @param interval 间隔时间(s)
 */
function createRepeat(fn, repeat, interval) {
	return async function (...args) {
		let index = 0

		let intervalId = setInterval(() => {
			index += 1

			if(index > repeat) {
				clearInterval(intervalId)
				intervalId = null
			} else {
				fn.apply(this, args)
			}
		}, interval * 1000)
	}
}

const fn = createRepeat(console.log, 3, 4)

fn('helloworld') // 每4秒输出一次helloWorld, 输出3次
